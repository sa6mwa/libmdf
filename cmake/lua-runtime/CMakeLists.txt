cmake_minimum_required(VERSION 3.24)
project(libmdf_lua_runtime C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
  message(FATAL_ERROR "Local Lua verification requires native x86_64 Linux")
endif()
set(lua_version 5.5.1)
set(digest 1c4b4068d67061f2a2231ad2b5422e77acea1487ea9890f6320af614f4373dce)
if(DEFINED CPKT_DEPENDENCY_CACHE)
  set(cache "${CPKT_DEPENDENCY_CACHE}")
elseif(DEFINED ENV{CPKT_DEPENDENCY_CACHE})
  set(cache "$ENV{CPKT_DEPENDENCY_CACHE}")
elseif(DEFINED ENV{XDG_CACHE_HOME})
  set(cache "$ENV{XDG_CACHE_HOME}/cpkt/deps")
else()
  set(cache "$ENV{HOME}/.cache/cpkt/deps")
endif()
set(CPKT_DEPENDENCY_CACHE "${cache}" CACHE PATH "Verified dependency archive cache")
set(archive_dir "${cache}/archives/sha256/${digest}")
file(MAKE_DIRECTORY "${archive_dir}" "${cache}/locks")
set(archive "${archive_dir}/lua-${lua_version}.tar.gz")
file(LOCK "${cache}/locks/${digest}.lock" GUARD PROCESS TIMEOUT 600)
set(actual "")
if(EXISTS "${archive}")
  file(SHA256 "${archive}" actual)
endif()
if(NOT actual STREQUAL digest)
  if(EXISTS "${archive}")
    message(FATAL_ERROR "Corrupt cached Lua archive: ${archive}; expected ${digest}")
  endif()
  file(DOWNLOAD "https://www.lua.org/ftp/lua-${lua_version}.tar.gz"
    "${archive}.tmp" TLS_VERIFY ON TIMEOUT 120 INACTIVITY_TIMEOUT 30 STATUS status)
  list(GET status 0 code)
  if(code EQUAL 0)
    file(SHA256 "${archive}.tmp" actual)
  endif()
  if(NOT code EQUAL 0 OR NOT actual STREQUAL digest)
    file(REMOVE "${archive}.tmp")
    message(FATAL_ERROR "Lua ${lua_version} download/checksum failed: ${status}; expected ${digest}")
  endif()
  file(RENAME "${archive}.tmp" "${archive}")
endif()
set(source "${CMAKE_CURRENT_BINARY_DIR}/source/lua-${lua_version}/src")
if(NOT EXISTS "${source}/lua.c")
  file(ARCHIVE_EXTRACT INPUT "${archive}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/source")
endif()
file(GLOB sources "${source}/*.c")
list(REMOVE_ITEM sources "${source}/lua.c" "${source}/luac.c")
add_library(lua_core STATIC ${sources})
target_compile_options(lua_core PRIVATE -O3)
target_compile_definitions(lua_core PRIVATE NDEBUG)
target_compile_definitions(lua_core PRIVATE LUA_USE_LINUX)
add_executable(lua "${source}/lua.c")
target_compile_options(lua PRIVATE -O3)
target_compile_definitions(lua PRIVATE LUA_USE_LINUX)
target_link_libraries(lua PRIVATE lua_core m dl)
target_link_options(lua PRIVATE -Wl,-E)
libmdf_configure_development_runtime(lua)
set_target_properties(lua PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
target_link_options(lua PRIVATE "-Wl,-rpath,${LIBMDF_LUA_MDF_LIBDIR}")
foreach(header lua.h luaconf.h lauxlib.h lualib.h)
  configure_file("${source}/${header}" "${CMAKE_CURRENT_BINARY_DIR}/include/${header}" COPYONLY)
endforeach()
