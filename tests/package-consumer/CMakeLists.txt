cmake_minimum_required(VERSION 3.24)
project(libmdf_package_consumer C)
find_package(libmdf CONFIG REQUIRED)
foreach(kind static shared)
  add_executable(consumer_${kind} main.c)
  target_link_libraries(consumer_${kind} PRIVATE libmdf::mdf_${kind})
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    libmdf_configure_development_runtime(consumer_${kind})
  endif()
endforeach()
find_package(PkgConfig REQUIRED)
pkg_check_modules(MDF REQUIRED IMPORTED_TARGET libmdf)
add_executable(consumer_pkgconfig main.c)
target_link_libraries(consumer_pkgconfig PRIVATE PkgConfig::MDF)
target_link_directories(consumer_pkgconfig PRIVATE ${MDF_LIBRARY_DIRS})
set_target_properties(consumer_pkgconfig PROPERTIES BUILD_RPATH "${MDF_LIBRARY_DIRS}")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  libmdf_configure_development_runtime(consumer_pkgconfig)
endif()
